def reap(self, ages):
    """Reap the records whose age exceeds ``window_s``.

    ``ages`` maps a record key to that record's age in seconds. Per
    docs/policy/03-retention.md, reaping changes a record's state to
    ``abandoned`` and returns its slot; the record itself stays in the
    stage and remains visible to ``snapshot()``. Returns the number of
    records reaped by this call; a key the stage has never seen is
    ignored, and an already-reaped record is not reaped again.
    """
    reaped = 0
    for key, age in ages.items():
        record = self._manifests.get(key)
        if record is None or age <= self.window_s:
            continue
        if record["state"] == "abandoned":
            continue
        record["state"] = "abandoned"
        reaped += 1
    return reaped

def active_count(self):
    """Count the records currently occupying a slot against ``limit``.

    Per docs/policy/03-retention.md, every record counts from acceptance
    until it reaches ``settled`` or ``abandoned``; only those two states
    are exempt.
    """
    return sum(
        1
        for record in self._manifests.values()
        if record["state"] not in ("settled", "abandoned")
    )
